\blue{\hand{\Large Theorem (Continuous Functional Calculus).}}

\blue{Let $A$ be a unital $C^*$-algebra and let $a\in A$ be normal,}
\[
  \red{a^*a=aa^*.}
\]
\blue{Then there exists a unique isometric $*$-homomorphism}
\[
  \blue{F_a:C(\sigma_A(a))\longrightarrow A,
  \qquad f\longmapsto f(a),}
\]
\blue{such that}
\[
\blue{F_a(\operatorname{id}_{\sigma(a)})=a,
\qquad F_a(1_{\sigma(a)})=1_A.}
\]
\blue{Moreover, its image is $C^*(a,1_A)$.}
\src{Statement: Section 2 notes, Theorem 2.22; the image statement is also recorded in the comprehensive study notes, \S3.4.}

\green{\textbf{Meaning of ``isometric''.}}
\[
\green{\|F_a(f)\|_A=\|f\|_\infty
=\max_{\lambda\in\sigma_A(a)}|f(\lambda)|.}
\]

\divider

\blue{\hand{Proof. Step 1: the commutative algebra generated by $a$.}}

Set
\[
  \blue{B:=C^*(a,1_A).}
\]
Because $a$ is normal, $a$ commutes with $a^*$; hence the $*$-algebra generated by
$a$ and $1_A$ is commutative, and so is its norm closure $B$.
\src{Tarea 1, Problem 3(d); cf. the comprehensive notes, \S4.3, ``Un normal genera una álgebra involutiva conmutativa''.}

By the commutative Gelfand--Naimark theorem,
\[
\blue{
\Lambda_B:B\longrightarrow C(\Omega(B)),
\qquad b\longmapsto\widehat b,
\qquad \widehat b(\varphi)=\varphi(b),
}
\]
is an isometric $*$-isomorphism.
\src{Section 2 notes, Theorem 2.20.}

\newpage

\blue{\hand{Step 2: identify the character space with the spectrum.}}

By Proposition 2.18,
\[
\blue{
\Phi:\Omega(B)\longrightarrow\sigma_A(a),
\qquad \Phi(\varphi)=\varphi(a),
}
\]
is a homeomorphism.
\src{Section 2 notes, Proposition 2.18. Spectral permanence used there is Proposition 2.17; the spectral description by characters ultimately comes from Part 1, Theorem 1.22.}

Define
\[
\blue{
\Phi_*:C(\Omega(B))\longrightarrow C(\sigma_A(a)),
\qquad \Phi_*(g):=g\circ\Phi^{-1}.
}
\]
It is plainly linear, multiplicative, unital, and preserves involution. It is isometric because
\begin{align*}
\|\Phi_*(g)\|_\infty
&=\sup_{\lambda\in\sigma_A(a)}
  \bigl|g(\Phi^{-1}(\lambda))\bigr|\\
&=\sup_{\varphi\in\Omega(B)}|g(\varphi)|
=\|g\|_\infty.
\end{align*}
It is surjective: if $f\in C(\sigma_A(a))$, set $g=f\circ\Phi$. Then
\[
  \Phi_*(g)=f\circ\Phi\circ\Phi^{-1}=f.
\]
Hence $\Phi_*$ is an isometric $*$-isomorphism.

\green{For later use, if $\lambda\in\sigma_A(a)$ and
$\varphi=\Phi^{-1}(\lambda)$, then}
\[
\green{\lambda=\Phi(\varphi)=\varphi(a).}
\]

\newpage

\blue{\hand{Step 3: compute the images of the generators.}}

Consider the composition
\[
\blue{
\Theta:=\Phi_*\circ\Lambda_B:
B\longrightarrow C(\sigma_A(a)).
}
\]
Since both factors are isometric $*$-isomorphisms, so is $\Theta$.

Let $\lambda\in\sigma_A(a)$ and $\varphi:=\Phi^{-1}(\lambda)$. Then
\begin{align*}
\Theta(1_A)(\lambda)
&=\Phi_*(\widehat{1_A})(\lambda)
 =\widehat{1_A}(\Phi^{-1}(\lambda))\\
&=\widehat{1_A}(\varphi)
 =\varphi(1_A)
 =1
 =1_{\sigma(a)}(\lambda),
\end{align*}
so
\[
\blue{\Theta(1_A)=1_{\sigma(a)}.}
\]
Likewise,
\begin{align*}
\Theta(a)(\lambda)
&=\Phi_*(\widehat a)(\lambda)
 =\widehat a(\Phi^{-1}(\lambda))\\
&=\widehat a(\varphi)
 =\varphi(a)
 =\lambda
 =\operatorname{id}_{\sigma(a)}(\lambda),
\end{align*}
so
\[
\blue{\Theta(a)=\operatorname{id}_{\sigma(a)}.}
\]

Therefore the inverse
\[
\blue{
F_a:=\Theta^{-1}:
C(\sigma_A(a))\longrightarrow B\subseteq A
}
\]
is an isometric $*$-isomorphism and satisfies
\[
\blue{
F_a(\operatorname{id}_{\sigma(a)})=a,
\qquad
F_a(1_{\sigma(a)})=1_A.
}
\]
This proves existence, and it also shows
\[
\blue{\operatorname{Im}F_a=B=C^*(a,1_A).}
\]

\newpage

\blue{\hand{Step 4: uniqueness.}}

Suppose
\[
G:C(\sigma_A(a))\longrightarrow A
\]
is another isometric $*$-homomorphism satisfying
\[
G(\operatorname{id}_{\sigma(a)})=a,
\qquad G(1_{\sigma(a)})=1_A.
\]
Because $G$ preserves the involution,
\[
G(\overline z)=G(z)^*=a^*.
\]
Thus $G$ and $F_a$ agree on every polynomial in the coordinate functions
$z$ and $\overline z$. Those polynomials form a unital $*$-subalgebra of
$C(\sigma_A(a))$ that separates points. By the complex Stone--Weierstrass theorem,
they are dense in $C(\sigma_A(a))$.
\src{Section 2 notes, Theorem 2.19 (Stone--Weierstrass).}

Both $G$ and $F_a$ are continuous (indeed isometric), so equality on the dense
subalgebra extends to all of $C(\sigma_A(a))$. Hence
\[
\blue{G=F_a.}
\]
This proves uniqueness.

\divider

\textcolor{sourcegray}{\textbf{Result map.}}
\begin{itemize}[leftmargin=1.7em,itemsep=0.2em,topsep=0.2em]
\item \textcolor{sourcegray}{Normality $\Rightarrow C^*(a,1_A)$ commutative: Tarea 1, Problem 3(d).}
\item \textcolor{sourcegray}{$\Omega(C^*(a,1_A))\cong\sigma_A(a)$: Section 2, Proposition 2.18.}
\item \textcolor{sourcegray}{Gelfand transform is an isometric $*$-isomorphism: Section 2, Theorem 2.20.}
\item \textcolor{sourcegray}{Density of polynomials in $z,\bar z$: Section 2, Theorem 2.19.}
\item \textcolor{sourcegray}{Continuous functional calculus: Section 2, Theorem 2.22.}
\end{itemize}
